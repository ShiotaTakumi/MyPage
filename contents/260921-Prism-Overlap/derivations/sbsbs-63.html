#!/usr/bin/env python3
"""SBSBS-63: the two end faces do not overlap for h_1(n) <= h <= h_2(n), where 25 <= n <= 25.

The family has l0 = 3, p1 = 4, d1 = -1, l1 = 2, p2 = 6, d2 = -1, l2 = 2 (pattern SBSBS). Its two end
faces are the bands S1 and S2, and the line is the one through the side p_3 p_4 of S1, opposite B1.
The vertices of the prism are b_j on B1 and t_j on B2, where the side face s_j has the vertices b_j,
b_{j+1}, t_{j+1}, t_j (the labels are taken modulo n), and alpha = 2 pi / n.

The path partial unfolding is laid down as on the page of the family:
  - B1 has the vertices b_j = sum_{i<j} (cos i alpha, sin i alpha), counterclockwise from
    b_0 = (0, 0) and b_1 = (1, 0);
  - the band S1 lies along the edge b_{p1} b_{p1+1} of B1, outside B1;
  - the band S0 = {s_0, ..., s_{l0-1}} lies below the edge b_0 b_1, along the x-axis;
  - B2 is joined below the last side face of S0, as a mirror image of B1;
  - the band S2 lies along the edge t_{p2} t_{p2+1} of B2, outside B2.
A band has the corners c_1, on its base at the side face s_{p_i}, c_2, on its base at the other end,
and c_3 = c_1 + h u and c_4 = c_2 + h u, where u is the unit normal to the edge of the base pointing
away from the base. The corners of S1 are p_1, ..., p_4 and those of S2 are q_1, ..., q_4.
For each corner q of S2, f_q(h) = (q - p_3) . (sin 4 alpha, -cos 4 alpha) is its signed distance
from the line, positive on the side away from S1. It has the form A + B h, where A and B depend on
alpha only; the program prints A and B, and then f_q(h_1(n)) and f_q(h_2(n)) for each n. Their signs
are decided by interval arithmetic (mpmath, 50 digits), which gives an interval that contains the
exact value; only if the interval contains 0, the value is tested for 0 exactly (by its minimal
polynomial), and, if it is not 0, computed again with twice as many digits until the interval does
not contain 0.

SymPy is needed (pip install sympy; mpmath comes with it). The program also runs in the browser,
with Pyodide, on the page of the family.
"""

import sympy as sp
from mpmath import iv
from sympy.simplify.fu import TR8

iv.dps = 50

a, h = sp.symbols("alpha h", positive=True)
vec = lambda x, y: sp.Matrix([x, y])
cos = lambda k: sp.cos(k * a)
sin = lambda k: sp.sin(k * a)

# The family and the line
L0, P1, D1, L1, P2, D2, L2 = 3, 4, -1, 2, 6, -1, 2
NS = range(25, 26)


# ------------------------------------------------------------------ the unfolding

def b(j):
    """The vertex b_j of B1."""
    if j >= 0:
        return vec(sum(cos(i) for i in range(j)), sum(sin(i) for i in range(j)))
    return -vec(sum(cos(i) for i in range(j, 0)), sum(sin(i) for i in range(j, 0)))


def turn(v, k):
    """The vector v turned by k alpha."""
    return vec(cos(k) * v[0] - sin(k) * v[1], sin(k) * v[0] + cos(k) * v[1])


def t(j):
    """The vertex t_j of B2. The last side face s_{l0-1} of S0 has its side on B2 from (l0 - 1, -h)
    to (l0, -h); B2 is B1 turned so that its edge b_{l0-1} b_{l0} is horizontal, reflected in the
    x-axis, and moved so that this edge lies on that side."""
    q = turn(b(j) - b(L0 - 1), -(L0 - 1))
    return vec(L0 - 1 + q[0], -h - q[1])


def band(corner, P, D, L, right):
    """The band of the L side faces s_P, s_{P+D}, ... that lies along the edge from corner(P) to
    corner(P + 1) of a base, outside the base: on the right of the edge if right, else on its left.
    Returns its corners [c_1, c_2, c_3, c_4], the unit vector along it from c_1 to c_2, and the unit
    normal to the edge pointing away from the base."""
    edge = corner(P + 1) - corner(P)                        # a unit vector
    out = vec(edge[1], -edge[0]) if right else vec(-edge[1], edge[0])
    start = corner(P) if D == 1 else corner(P + 1)          # c_1, at the side face s_P
    along = D * edge
    return [start, start + L * along, start + h * out, start + L * along + h * out], along, out


p, along1, out1 = band(b, P1, D1, L1, right=True)       # S1; B1 is counterclockwise
q, along2, out2 = band(t, P2, D2, L2, right=False)      # S2; B2 is clockwise


# ------------------------------------------------------------------ the signed distances

normal = out1                            # the unit normal to the side p_3 p_4 of S1 pointing away from S1
f = [sp.expand((v - p[2]).dot(normal)) for v in q]
A = [sp.expand(TR8(x.coeff(h, 0))) for x in f]      # product-to-sum
B = [sp.expand(TR8(x.coeff(h, 1))) for x in f]


# ------------------------------------------------------------------ the two ends of the interval

def h1(n):
    """h_1(n) (Section 2 of the home page), in alpha."""
    if n <= 12:
        return (sin(3) - sin(2)) / cos(1)
    if n <= 16:
        return (sin(3) - sin(2)) / (cos(1) - cos(3))
    if n <= 19:
        return (cos(4) + 2 - cos(1) - cos(2)) / sin(3)
    return (2 * sin(5) - sin(4) - sin(3) + sin(1)) / (cos(2) - cos(5))


def h2(n):
    """h_2(n) (Section 2 of the home page), in alpha."""
    return cos(1) / (sin(3) - sin(2)) if n == 11 else (cos(1) - cos(3)) / (sin(3) - sin(2))


def at(expr, n):
    """expr at alpha = 2 pi / n, as an interval that contains its exact value. Every number in expr
    becomes an interval: sin, cos, pi and the fractions (written mpf(p)/mpf(q)) are those of iv, and so
    is an integer, which expr may reduce to."""
    names = {"sin": iv.sin, "cos": iv.cos, "pi": iv.pi, "mpf": iv.mpf}
    return iv.mpf(sp.lambdify(a, expr, modules=[names, "mpmath"])(2 * iv.pi / n))


def sign(expr, n):
    """The sign of expr at alpha = 2 pi / n, decided as described at the top."""
    digits = iv.dps
    try:
        while True:
            x = at(expr, n)
            if x.a > 0:
                return 1
            if x.b < 0:
                return -1
            if iv.dps == digits:                        # only once: is the value exactly 0?
                X = sp.Symbol("X")
                if sp.minimal_polynomial(expr.subs(a, 2 * sp.pi / n), X) == X:
                    return 0
            iv.dps *= 2
    finally:
        iv.dps = digits


def values(n):
    """[(h, [f_q(h) for the four corners q])] at h = h_1(n) and h = h_2(n), as intervals."""
    out = []
    for H in (h1(n), h2(n)):
        out.append((at(H, n), [at(A[k] + B[k] * H, n) for k in range(4)]))
    return out


def main():
    print("f_q(h) = A + B h for the corners q_1, ..., q_4 of S2:")
    for k in range(4):
        print(f"  q_{k + 1}:  A = {A[k]}")
        print(f"         B = {B[k]}")
    positive = True
    for side, name in enumerate(("h_1(n)", "h_2(n)")):
        print(f"\nAt h = {name}:")
        print("   n         h      f(q_1)     f(q_2)     f(q_3)     f(q_4)")
        for n in NS:
            H, fs = values(n)[side]
            end = (h1, h2)[side](n)
            positive &= all(sign(A[k] + B[k] * end, n) == 1 for k in range(4))
            print(f"  {n:2d}  {float(H.mid):8.6f}  " + "  ".join(f"{float(x.mid):9.6f}" for x in fs))
    print("\nEvery value is positive." if positive else "\nSome value is not shown to be positive.")


if __name__ == "__main__":
    main()

#!/usr/bin/env wolframscript
(* ::Package:: *)

(* SBSBS-63: the two end faces do not overlap for h_1(n) <= h <= h_2(n), where 25 <= n <= 25.

   Mathematica version of sbsbs-63.py, written separately from it. The family has l0 = 3, p1 = 4,
   d1 = -1, l1 = 2, p2 = 6, d2 = -1, l2 = 2, and the line is the one through the side p_3 p_4 of S1,
   opposite B1. The script lays the path partial unfolding down in the coordinates of the page (B2
   is walked along its edges here, instead of being obtained as a mirror image of B1, and each band
   is walked along its base one side face at a time), checks that the signed distances f_q(h) of the
   four corners q of S2 from the line equal the formulas on the page, and that f_q(h_1(n)) > 0 and
   f_q(h_2(n)) > 0 for every n. Identities in alpha are verified as (left side) - (right side) = 0.
   The sign of a number for a particular n (alpha = 2 Pi/n) is decided by interval arithmetic, and,
   only if the interval contains 0, exactly, as the sign of the algebraic number that RootReduce
   gives.

   Run:  wolframscript -file sbsbs-63.wls *)

$Assumptions = a > 0 && h > 0;

passed = 0; failed = 0;
(* zero as an identity in alpha: after TrigReduce (products to sums), else by FullSimplify *)
zeroQ[e_List] := AllTrue[Flatten[e], zeroQ];
zeroQ[e_] := TrigReduce[Expand[e]] === 0 || FullSimplify[e] === 0;
check[label_String, e_] := If[zeroQ[e],
  passed++; Print[label, "  OK"],
  failed++; Print[label, "  NG: ", InputForm[FullSimplify[e]]]];
checkTrue[label_String, b_] := If[TrueQ[b],
  passed++; Print[label, "  OK"],
  failed++; Print[label, "  NG: ", b]];

(* the sign of x, an expression in a, at a = 2 Pi/n. First by interval arithmetic: a is replaced by an
   interval of 80 digits that contains 2 Pi/n, and Mathematica computes an interval that contains the
   value, rounding outward. Only if this interval contains 0 is the sign decided exactly: RootReduce
   gives the value as an algebraic number (a Root object with an isolating interval), whose sign
   Mathematica decides exactly. exact counts these. *)
exact = 0;
around[n_] := With[{c = N[2 Pi/n, 80]}, Interval[{c - 10^-70, c + 10^-70}]];
sgn[x_, n_] := With[{v = x /. a -> around[n]},
  Which[Min[v] > 0, 1, Max[v] < 0, -1, True, exact++; Sign[RootReduce[x /. a -> 2 Pi/n]]]];
positiveQ[x_, n_] := sgn[x, n] === 1;

(* ------------------------------------------------------------ the family *)

{l0, p1, d1, l1, p2, d2, l2} = {3, 4, -1, 2, 6, -1, 2};
ns = Range[25, 25];

(* ------------------------------------------------------------ the unfolding *)

(* B1: b_0 = (0, 0), b_1 = (1, 0), counterclockwise; the edge b_j b_{j+1} has the direction j a *)
b[j_] := Sum[{Cos[i a], Sin[i a]}, {i, 0, j - 1}];

(* S0 lies below b_0 b_1 along the x-axis, and B2 below its last side face, from t_{l0-1} = (l0 - 1, -h)
   to t_{l0} = (l0, -h). B2 is walked clockwise from there: its edge t_j t_{j+1} has the direction
   -(j - (l0 - 1)) a *)
dir2[j_] := {Cos[(j - (l0 - 1)) a], -Sin[(j - (l0 - 1)) a]};
t[j_] := {l0 - 1, -h} + Sum[dir2[i], {i, l0 - 1, j - 1}];

(* a band of l side faces s_p, s_(p+d), ... along an edge of a base with the direction dir, outside the
   base: from its corner c_1 on the base at the side face s_p, it is walked along the base one side
   face (the step d dir) at a time, to its corner c_2 at the other end; c_3 and c_4 are c_1 and c_2
   moved by h in the direction out, away from the base *)
band[c1_, dir_, d_, l_, out_] := With[{c2 = c1 + Sum[d dir, {k, 1, l}]}, {c1, c2, c1 + h out, c2 + h out}];

(* S1 lies along the edge b_{p1} b_{p1+1}, outside B1: B1 is on the left of the edge, as it is walked
   counterclockwise, so the outward normal is the edge turned by -90 degrees *)
dir1 = {Cos[p1 a], Sin[p1 a]};
out1 = {dir1[[2]], -dir1[[1]]};
s1 = band[If[d1 == 1, b[p1], b[p1 + 1]], dir1, d1, l1, out1];

(* S2 lies along the edge t_{p2} t_{p2+1}, outside B2: B2 is on the right of the edge, as it is walked
   clockwise, so the outward normal is the edge turned by +90 degrees *)
out2 = {-dir2[p2][[2]], dir2[p2][[1]]};
s2 = band[If[d2 == 1, t[p2], t[p2 + 1]], dir2[p2], d2, l2, out2];

(* the corners of S1 are p_1, ..., p_4 and those of S2 are q_1, ..., q_4 *)
p = s1; q = s2;

(* the signed distance from the line through the side p_3 p_4 of S1, positive away from S1 *)
normal = out1;
f = Expand[(# - p[[3]]) . normal] & /@ q;

(* ------------------------------------------------------------ the formulas on the page *)

page = {
  h (Cos[4 a] - 1) - Sin[a] - Sin[2 a] - Sin[3 a] + 2 Sin[4 a] + Sin[5 a] + Sin[6 a] + Sin[7 a] + Sin[8 a],
  h (Cos[4 a] - 1) - Sin[a] - Sin[2 a] - Sin[3 a] + 2 Sin[4 a] + Sin[5 a] + Sin[6 a] + Sin[7 a] - Sin[8 a],
  h (Cos[4 a] - Cos[8 a] - 1) - Sin[a] - Sin[2 a] - Sin[3 a] + 2 Sin[4 a] + Sin[5 a] + Sin[6 a] + Sin[7 a] + Sin[8 a],
  h (Cos[4 a] - Cos[8 a] - 1) - Sin[a] - Sin[2 a] - Sin[3 a] + 2 Sin[4 a] + Sin[5 a] + Sin[6 a] + Sin[7 a] - Sin[8 a]};
Do[check["f(q_" <> ToString[k] <> ")", f[[k]] - page[[k]]], {k, 4}];
check["S1 lies in f <= 0: f(p_1) = -h", (p[[1]] - p[[3]]) . normal - (-h)];

(* ------------------------------------------------------------ the two ends of the interval *)

(* h_1(n) and h_2(n) (Section 2 of the home page), in a *)
h1[n_] := Which[
  n <= 12, (Sin[3 a] - Sin[2 a])/Cos[a],
  n <= 16, (Sin[3 a] - Sin[2 a])/(Cos[a] - Cos[3 a]),
  n <= 19, (Cos[4 a] + 2 - Cos[a] - Cos[2 a])/Sin[3 a],
  True, (2 Sin[5 a] - Sin[4 a] - Sin[3 a] + Sin[a])/(Cos[2 a] - Cos[5 a])];
h2[n_] := If[n == 11, Cos[a]/(Sin[3 a] - Sin[2 a]), (Cos[a] - Cos[3 a])/(Sin[3 a] - Sin[2 a])];

(* f_q(h_side(n)), in a *)
value[k_, side_, n_] := page[[k]] /. h -> If[side == 1, h1[n], h2[n]];
Do[checkTrue["f(q_" <> ToString[k] <> ")(h_" <> ToString[side] <> "(n)) > 0 for 25 <= n <= 25",
    AllTrue[ns, positiveQ[value[k, side, #], #] &]],
  {side, 2}, {k, 4}];

(* the table: f_q(h_side(n)) rounded to six decimals *)
Do[Print["n=", n, ", h=h_", side, "(n): ", StringRiffle[
    Table[ToString[NumberForm[N[value[k, side, n] /. a -> 2 Pi/n, 20], {12, 6}]], {k, 4}], "  "]],
  {side, 2}, {n, ns}];

Print["passed ", passed, ", failed ", failed, "; signs decided exactly (RootReduce): ", exact];
Exit[If[failed == 0, 0, 1]];
